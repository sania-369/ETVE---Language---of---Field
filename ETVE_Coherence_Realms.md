import numpy as np

class InfinitumDemonstrator:
    """
    Демонстратор логики бесконечности.
    Показывает, как вариация топологии (числа скрытых измерений, Z-поправки)
    порождает бесконечный спектр возможных вселенных.
    """
    def __init__(self):
        # Фундаментальный базис — неизменен для всех вселенных
        self.Phi = (1.0 + np.sqrt(5.0)) / 2.0
        self.pi = np.pi
        self.Z_res = np.sqrt(3.0)
        
        # Параметры НАШЕЙ вселенной (эталон)
        self.our_hidden_dims = 7      # Скрытых измерений
        self.our_unfolded_dims = 4    # Развёрнутых измерений
        
    def calculate_universe(self, hidden_dims, z_amplitude=None):
        """
        Вычисляет фундаментальные константы для вселенной с заданной топологией.
        
        Аргументы:
        - hidden_dims: число скрытых измерений (для нашей — 7)
        - z_amplitude: амплитуда Z-дыхания (для нашей — 1.0, т.е. √3 как есть)
        
        Возвращает:
        dict с константами этой вселенной
        """
        if z_amplitude is None:
            z_amplitude = 1.0
            
        Z_effective = self.Z_res * z_amplitude
        
        # 1. Калибровочный множитель α⁻¹
        # Формула: √(π × Φ³) + Z / 2^dims
        si_cal = np.sqrt(self.pi * (self.Phi ** 3)) + Z_effective / (2 ** hidden_dims)
        
        # 2. Топологическое ядро α⁻¹ (не зависит от измерений)
        pure_alpha = (self.pi * self.Phi**4 + self.pi**2 * self.Phi - 
                      1.0 / (self.Phi**3 * self.pi))
        
        # 3. Постоянная тонкой структуры
        alpha_inv = pure_alpha * si_cal
        
        # 4. Энергетический масштаб
        # Формула: 2^(3×unfolded) − Z⁴ × π³
        si_energy = (2 ** (3 * self.our_unfolded_dims) - 
                     (Z_effective ** 4) * (self.pi ** 3))
        
        # 5. Масса электрона (упрощённо)
        # Берём коэффициент пропорциональности из нашей вселенной и масштабируем
        # отношением калибровочных множителей
        our_si_energy = (2 ** (3 * self.our_unfolded_dims) - 
                         (self.Z_res ** 4) * (self.pi ** 3))
        energy_ratio = si_energy / our_si_energy
        
        # 6. Гравитационный масштаб
        grav_denom = ((self.Phi ** 20) * 2.0 * (self.pi ** 2) + 
                      (self.pi ** 5) * si_cal)
        si_gravity = 1.0 / grav_denom
        
        # 7. Оценка когерентности этой вселенной
        # C тем выше, чем ближе hidden_dims к 7 (оптимальная топология)
        C_estimate = 1.0 / (1.0 + abs(hidden_dims - 7) * 0.1)
        
        return {
            "hidden_dimensions": hidden_dims,
            "z_amplitude": z_amplitude,
            "alpha_inverse": alpha_inv,
            "electron_mass_ratio": energy_ratio,
            "gravity_scale": si_gravity,
            "coherence_estimate": C_estimate,
            "is_stable": 0.8 < C_estimate < 1.0
        }
    
    def scan_multiverse(self, dims_range=(0, 20), z_range=(0.5, 2.0), samples=10):
        """
        Сканирует спектр возможных вселенных, варьируя топологию.
        """
        results = []
        
        for hidden_dims in range(dims_range[0], dims_range[1] + 1):
            for z_amp in np.linspace(z_range[0], z_range[1], samples):
                universe = self.calculate_universe(hidden_dims, z_amp)
                results.append(universe)
              return results
    
    def demonstrate(self):

        print("[ВАРИАТИВНОСТЬ]: Топология (число измерений) и амплитуда Z-дыхания.")
        print("[СЛЕДСТВИЕ]: Бесконечный спектр физических констант.")

        print(f"   1/α = {our['alpha_inverse']:.6f}")
        print(f"   Когерентность C ≈ {our['coherence_estimate']:.4f}")
        print(f"   Стабильна: {'ДА ✅' if our['is_stable'] else 'НЕТ ❌'}")
        
        # Соседние топологии
        print("\n🔶 СОСЕДНИЕ ВСЕЛЕННЫЕ (вариация скрытых измерений):")
        print(f"   {'Dims':<6} {'1/α':<12} {'C':<8} {'Стабильна'}")
        print(f"   {'-'*40}")

        print("-" * 75)
        print("1. Базис (Φ, π, √3) един и неизменен для всех вселенных.")
